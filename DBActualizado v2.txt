-- Eliminar la base de datos si ya existe
DROP DATABASE IF EXISTS fitforge_db;

-- Crear la base de datos
CREATE DATABASE fitforge_db;

-- Usar la base de datos
USE fitforge_db;

-- Eliminar el usuario si ya existe
DROP USER IF EXISTS 'fitforge_user'@'localhost';

-- Crear el usuario para la base de datos
CREATE USER 'fitforge_user'@'localhost' IDENTIFIED BY 'FitforgeClave123';
GRANT ALL PRIVILEGES ON fitforge_db.* TO 'fitforge_user'@'localhost';
FLUSH PRIVILEGES;

-- ==========================
-- 📌 TABLA DE CATEGORÍAS
-- ==========================
CREATE TABLE categoria (
  id_categoria INT NOT NULL AUTO_INCREMENT,
  descripcion VARCHAR(50) NOT NULL,
  activo BOOLEAN DEFAULT TRUE,
  PRIMARY KEY (id_categoria)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE PRODUCTOS
-- ==========================
CREATE TABLE producto (
  id_producto INT NOT NULL AUTO_INCREMENT,
  id_categoria INT NOT NULL,
  descripcion VARCHAR(100) NOT NULL,
  detalle TEXT NOT NULL,
  precio DOUBLE NOT NULL,
  imagen VARCHAR(255), -- Nueva columna para almacenar la ruta de la imagen
  activo BOOLEAN DEFAULT TRUE,
  PRIMARY KEY (id_producto),
  FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE COLORES
-- ==========================
CREATE TABLE color (
  id_color INT NOT NULL AUTO_INCREMENT,
  nombre_color VARCHAR(50) NOT NULL,
  PRIMARY KEY (id_color)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE TALLAS
-- ==========================
CREATE TABLE talla (
  id_talla INT NOT NULL AUTO_INCREMENT,
  nombre_talla VARCHAR(50) NOT NULL,
  PRIMARY KEY (id_talla)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE PRODUCTO_COLOR_TALLA
-- ==========================
CREATE TABLE producto_color_talla (
  id_producto_color_talla INT NOT NULL AUTO_INCREMENT,
  id_producto INT NOT NULL,
  id_color INT NULL, -- Permite NULL
  id_talla INT NULL, -- Permite NULL
  existencias INT NOT NULL,
  PRIMARY KEY (id_producto_color_talla),
  FOREIGN KEY (id_producto) REFERENCES producto(id_producto),
  FOREIGN KEY (id_color) REFERENCES color(id_color),
  FOREIGN KEY (id_talla) REFERENCES talla(id_talla)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE USUARIOS
-- ==========================
CREATE TABLE usuario (
  id_usuario INT NOT NULL AUTO_INCREMENT,
  username VARCHAR(20) NOT NULL,
  password VARCHAR(512) NOT NULL,
  nombre VARCHAR(50) NOT NULL,
  correo VARCHAR(100) NOT NULL,
  telefono VARCHAR(15),
  avatar VARCHAR(255), -- Nueva columna para almacenar la imagen de perfil
  activo BOOLEAN DEFAULT TRUE,
  PRIMARY KEY (id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE FACTURAS
-- ==========================
CREATE TABLE factura (
  id_factura INT NOT NULL AUTO_INCREMENT,
  id_usuario INT NOT NULL,
  fecha DATE NOT NULL,
  total DOUBLE NOT NULL,
  estado ENUM('Pendiente', 'Pagada', 'Cancelada') DEFAULT 'Pendiente',
  PRIMARY KEY (id_factura),
  FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE VENTAS
-- ==========================
CREATE TABLE venta (
  id_venta INT NOT NULL AUTO_INCREMENT,
  id_factura INT NOT NULL,
  id_producto INT NOT NULL,
  precio DOUBLE NOT NULL,
  cantidad INT NOT NULL,
  PRIMARY KEY (id_venta),
  FOREIGN KEY (id_factura) REFERENCES factura(id_factura),
  FOREIGN KEY (id_producto) REFERENCES producto(id_producto)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE ROLES
-- ==========================
CREATE TABLE rol (
  id_rol INT NOT NULL AUTO_INCREMENT,
  nombre VARCHAR(20) NOT NULL,
  id_usuario INT NOT NULL,
  PRIMARY KEY (id_rol),
  FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE PERMISOS (SECURITY)
-- ==========================
CREATE TABLE request_matcher (
  id_request_matcher INT AUTO_INCREMENT NOT NULL,
  pattern VARCHAR(255) NOT NULL,
  role_name VARCHAR(50) NOT NULL,
  PRIMARY KEY (id_request_matcher)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ==========================
-- 📌 TABLA DE CONTACTO
-- ==========================
CREATE TABLE contacto (
  id_contacto INT NOT NULL AUTO_INCREMENT,
  nombre VARCHAR(50) NOT NULL,
  apellido VARCHAR(50) NOT NULL,
  correo VARCHAR(100) NOT NULL,
  mensaje VARCHAR(255),
  activo BOOLEAN DEFAULT TRUE,
  PRIMARY KEY (id_contacto)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
